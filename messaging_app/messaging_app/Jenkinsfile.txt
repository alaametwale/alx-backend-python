pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                // سحب الكود من GitHub مع تحديد الفرع
                git branch: 'main', url: 'https://github.com/alaametwale/alx-backend-python.git'
            }
        }

        stage('Install Dependencies') {
            steps {
                // تثبيت dependencies من ملف requirements.txt
                sh 'pip3 install -r messaging_app/requirements.txt'
            }
        }

        stage('Run Tests') {
            steps {
                // تشغيل pytest مع إنشاء تقرير XML
                sh 'pytest messaging_app/messaging_app/tests --junitxml=report.xml'
            }
        }

        stage('Archive Report') {
            steps {
                // حفظ التقرير كأرشيف
                archiveArtifacts artifacts: 'report.xml', allowEmptyArchive: true
            }
        }
    }
}
